package io.rivomesh.platform;

import java.util.*;

/** Durable inputs bind an explicit reviewed plan; actor identity always comes from authentication. */
public final class RepairRequest {
    public String reportId,planId,approvedHash;
    public Set<String> scope=new TreeSet<>();
    void validate(boolean execute){
        if(reportId==null||!UUID.fromString(reportId).toString().equals(reportId)||scope==null)throw new IllegalArgumentException("REPAIR_REQUEST_INVALID");
        if(execute){if(planId==null||!UUID.fromString(planId).toString().equals(planId)||approvedHash==null||!approvedHash.matches("[0-9a-f]{64}")||!scope.isEmpty())throw new IllegalArgumentException("REPAIR_APPROVAL_INVALID");}
        else if(planId!=null||approvedHash!=null||scope.isEmpty()||scope.size()>200||scope.stream().anyMatch(s->s==null||!s.matches("[A-Za-z0-9][A-Za-z0-9._-]{0,127}/[0-9a-f]{64}")))throw new IllegalArgumentException("REPAIR_PLAN_SCOPE_INVALID");
        scope=new TreeSet<>(scope);
    }
}
