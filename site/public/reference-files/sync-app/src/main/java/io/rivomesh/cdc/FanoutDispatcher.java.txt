package io.rivomesh.cdc;

import java.sql.SQLException;
import java.util.*;

/** Subscriber failures are isolated. A durable sink may lead its journal ACK by one uncertain commit. */
public final class FanoutDispatcher {
    public interface Transformer { SourceTransaction normalize(Journal.Record record)throws Exception; }
    private final Journal journal;
    private final Journal.Lease lease;
    private final List<SinkAdapter> sinks;
    private final Map<String,TableSchema> schemas;
    private final Transformer transformer;
    private final Map<String,Long> retryAfter=new HashMap<>();
    private final Map<String,String> states=new LinkedHashMap<>();
    public FanoutDispatcher(Journal journal,Journal.Lease lease,List<SinkAdapter> sinks,Map<String,TableSchema> schemas,Transformer transformer){
        this.journal=journal;this.lease=lease;this.sinks=new ArrayList<>(sinks);this.schemas=schemas;this.transformer=sourceTransformer(transformer);
    }
    private SourceTransaction project(SinkAdapter sink,SourceTransaction normalized){
        SourceTransaction tx=Codec.decode(Codec.encode(normalized),SourceTransaction.class);List<Change> changes=new ArrayList<>();
        for(Change change:tx.changes){if(change.operation==Change.Operation.BARRIER){changes.add(change);continue;}TableSchema table=schemas.get(change.tableId);if(table==null)throw new IllegalStateException("HISTORICAL_SCHEMA_MISSING");if(sink.includes(table))changes.add(change);}
        tx.changes=changes;return tx;
    }
    /** Bounded serial helper; the runner gives each destination its own dispatcher thread. */
    public int drain(int maximumPerSink)throws Exception{
        if(Thread.currentThread().isInterrupted())throw new InterruptedException("SUBSCRIBER_STOPPED");
        if(maximumPerSink<1||maximumPerSink>1000)throw new IllegalArgumentException("DRAIN_LIMIT_INVALID");int applied=0;
        Map<String,Journal.Subscription> subscriptions=new HashMap<>();for(Journal.Subscription sub:journal.subscriptions())subscriptions.put(sub.id,sub);
        for(SinkAdapter sink:sinks){
            Journal.Subscription sub=subscriptions.get(sink.id());if(sub==null)throw new IllegalStateException("SUBSCRIPTION_NOT_FOUND");
            if(!"RUNNING".equals(sub.state)){states.put(sink.id(),"PAUSED");continue;}
            if(retryAfter.containsKey(sink.id())&&System.nanoTime()-retryAfter.get(sink.id())<0){states.put(sink.id(),"RETRYING");continue;}
            try{
                SinkAdapter.Checkpoint checkpoint=sink.activate(lease.epoch);
                if(checkpoint.sequence<sub.acknowledged||checkpoint.sequence>Math.addExact(sub.acknowledged,1))throw new IllegalStateException("SINK_JOURNAL_CHECKPOINT_DIVERGED");
                List<Journal.Record> records=read(sink.id(),maximumPerSink);int start=0;
                if(checkpoint.sequence>sub.acknowledged){
                    if(records.isEmpty()||records.get(0).sequence!=checkpoint.sequence)throw new IllegalStateException("COMMITTED_SINK_HISTORY_MISSING");
                    SourceTransaction tx=project(sink,transformer.normalize(records.get(0)));
                    if(!tx.id.equals(checkpoint.transactionId)||!Codec.sha256(Codec.encode(tx)).equals(checkpoint.transactionHash))throw new IllegalStateException("COMMITTED_SINK_CONTENT_MISMATCH");
                    acknowledge(sink.id(),checkpoint.sequence);start=1;applied++;
                }
                for(int i=start;i<records.size();i++){if(Thread.currentThread().isInterrupted())throw new InterruptedException("SUBSCRIBER_STOPPED");Journal.Record record=records.get(i);SourceTransaction tx=project(sink,transformer.normalize(record));sink.apply(lease.epoch,record.sequence,tx);acknowledge(sink.id(),record.sequence);applied++;}
                if(sub.errorCode!=null||retryAfter.containsKey(sink.id()))retrying(sink.id(),null);
                states.put(sink.id(),"RUNNING");retryAfter.remove(sink.id());
            }catch(JdbcJournal.Fenced failure){throw failure;}
            catch(InterruptedException stopped){throw stopped;}
            catch(Exception failure){
                // Source normalization faults stop the whole stream; target faults only pause that subscriber.
                // Transformer failures are tagged below and must never be mistaken for a target SQL failure.
                if(failure instanceof NormalizationFailure||failure instanceof JournalFailure)throw failure;
                String code=errorCode(failure);if(retryable(failure)){if(code.equals("TARGET_APPLY_ERROR"))code="TARGET_CONNECTION_RETRY";if(!"RETRYING".equals(states.get(sink.id()))||!Objects.equals(sub.errorCode,code))io.rivomesh.FailureDiagnostics.targetRetrying(sink.id(),code,failure);retrying(sink.id(),code);retryAfter.put(sink.id(),System.nanoTime()+java.util.concurrent.TimeUnit.SECONDS.toNanos(2));states.put(sink.id(),"RETRYING");}
                else{journal.pause(lease,sink.id(),code);states.put(sink.id(),"PAUSED");io.rivomesh.FailureDiagnostics.targetPaused(sink.id(),code,failure);}
            }
        }
        return applied;
    }
    public static final class NormalizationFailure extends Exception { public NormalizationFailure(Exception cause){super("SOURCE_NORMALIZATION_FAILED",cause);} }
    public static final class JournalFailure extends Exception { public JournalFailure(Exception cause){super("JOURNAL_OPERATION_FAILED",cause);} }
    private List<Journal.Record> read(String subscription,int limit)throws Exception{try{return journal.read(subscription,limit);}catch(JdbcJournal.Fenced failure){throw failure;}catch(Exception failure){throw new JournalFailure(failure);}}
    private void acknowledge(String subscription,long sequence)throws Exception{try{journal.acknowledge(lease,subscription,sequence);}catch(JdbcJournal.Fenced failure){throw failure;}catch(Exception failure){throw new JournalFailure(failure);}}
    private void retrying(String subscription,String code)throws Exception{try{journal.retrying(lease,subscription,code);}catch(JdbcJournal.Fenced failure){throw failure;}catch(Exception failure){throw new JournalFailure(failure);}}
    public static Transformer sourceTransformer(Transformer transformer){return record->{try{return transformer.normalize(record);}catch(Exception failure){if(failure instanceof JdbcJournal.Fenced)throw failure;throw new NormalizationFailure(failure);}};}
    private static boolean retryable(Exception failure){return io.rivomesh.Sql.retryable(failure);}
    private static String errorCode(Exception failure){if(failure instanceof io.rivomesh.SyncFailure)return ((io.rivomesh.SyncFailure)failure).code;if(failure instanceof SQLException)return io.rivomesh.Sql.targetErrorCode((SQLException)failure);String message=failure.getMessage();return message!=null&&message.matches("[A-Z][A-Z0-9_]{1,120}")?message:"TARGET_APPLY_ERROR";}
    public Map<String,String> states(){return new LinkedHashMap<>(states);}
}
